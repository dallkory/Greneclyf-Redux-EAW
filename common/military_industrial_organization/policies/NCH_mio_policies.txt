

# Bugsmere Industrial Concern
NCH_mio_policy_state_construction_board = {
	name = NCH_mio_policy_state_construction_board
	icon = GFX_mio_policy_vertical_integration
	allowed = {
		is_military_industrial_organization = NCH_bugsmere_industry_manufacturer_organization
	}
	available = { has_mio_size > 5 }
	on_add = {
		custom_effect_tooltip = NCH_mio_policy_state_construction_board_tt
		hidden_effect = {
			FROM = {
				add_to_variable = { var = NCH_industry_mio_construction_speed value = 0.05 }
				add_to_variable = { var = NCH_industry_mio_infrastructure_construction value = 0.10 }
				add_dynamic_modifier = { modifier = NCH_industry_mio_dynamic_modifier }
			}
		}
	}
	on_remove = {
		hidden_effect = {
			FROM = {
				add_to_variable = { var = NCH_industry_mio_construction_speed value = -0.05 }
				add_to_variable = { var = NCH_industry_mio_infrastructure_construction value = -0.10 }
			}
		}
	}
	ai_will_do = { factor = 1 }
}
